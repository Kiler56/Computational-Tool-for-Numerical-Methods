Begin

Define function approx_derivative(f, x):
    h ← 1e-6
    return (f(x + h) - f(x - h)) / (2*h)

Define function newton(f, x0, tolerance, max_iter):
    x ← x0

    For i from 1 to max_iter:
        fx ← f(x)
        dfx ← approx_derivative(f, x)

        If dfx = 0:
            Print "Error: zero derivative"
            Return None

        x_new ← x - fx / dfx

        If |x_new - x| < tolerance:
            Print "Converged in i iterations"
            Return x_new

        x ← x_new

    Print "Did not converge"
    Return None

End
