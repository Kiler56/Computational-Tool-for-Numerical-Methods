Begin

Define function approx_derivative(f, x):
    h ← 1e-6
    return (f(x + h) - f(x - h)) / (2*h)

Define function approx_second_derivative(f, x):
    h ← 1e-6
    return (f(x + h) - 2*f(x) + f(x - h)) / (h*h)

Define function multiple_roots(f, x0, tolerance, max_iter):
    x ← x0

    For i from 1 to max_iter:
        fx ← f(x)
        dfx ← approx_derivative(f, x)
        ddfx ← approx_second_derivative(f, x)
        denom ← dfx*dfx - fx*ddfx

        If denom = 0:
            Print "Error: division by zero"
            Return None

        x_new ← x - (fx * dfx) / denom

        If |x_new - x| < tolerance:
            Print "Converged in i iterations"
            Return x_new

        x ← x_new

    Print "Did not converge"
    Return None

End
